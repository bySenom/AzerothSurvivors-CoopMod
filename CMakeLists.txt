cmake_minimum_required(VERSION 3.24)
project(AzerothCoop VERSION 0.1.0 LANGUAGES C CXX)
enable_language(ASM_MASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The original game and this module require Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(azs_coop SHARED src/module.cpp)
target_link_libraries(azs_coop PRIVATE minhook)
target_compile_definitions(azs_coop PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_library(azs_transport SHARED src/transport.cpp src/rust_pair.asm)
target_link_libraries(azs_transport PRIVATE minhook ws2_32 bcrypt)
target_compile_definitions(azs_transport PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_executable(AzerothCoop src/launcher.cpp)
add_library(azs_gameplay SHARED src/gameplay.cpp)
target_link_libraries(azs_gameplay PRIVATE minhook)
target_compile_definitions(azs_gameplay PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_library(azs_lobby SHARED src/lobby.cpp)
target_link_libraries(azs_lobby PRIVATE minhook ws2_32 bcrypt user32 gdi32)
target_compile_definitions(azs_lobby PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(AzerothCoop PRIVATE bcrypt)
target_compile_definitions(AzerothCoop PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MSVC)
  target_compile_options(azs_coop PRIVATE /W4 /EHsc)
  target_compile_options(AzerothCoop PRIVATE /W4 /EHsc)
  target_compile_options(azs_gameplay PRIVATE /W4 /EHsc)
  target_compile_options(azs_lobby PRIVATE /W4 /EHsc /utf-8)
  target_compile_options(azs_transport PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endif()
foreach(target azs_coop azs_transport azs_gameplay azs_lobby AzerothCoop)
  if(MSVC)
    target_link_options(${target} PRIVATE "/PDBALTPATH:%_PDB%")
  endif()
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()

# Keep the previous inspected native ABI in separate modules. The launcher
# selects this directory only for that adapter's exact executable hash.
foreach(target azs_coop azs_transport azs_gameplay azs_lobby AzerothCoop)
  get_target_property(sources ${target} SOURCES)
  get_target_property(kind ${target} TYPE)
  set(legacy "${target}_legacy")
  if(kind STREQUAL "EXECUTABLE")
    add_executable(${legacy} ${sources})
  else()
    add_library(${legacy} SHARED ${sources})
  endif()
  get_target_property(libraries ${target} LINK_LIBRARIES)
  target_link_libraries(${legacy} PRIVATE ${libraries})
  target_compile_definitions(${legacy} PRIVATE AZS_LEGACY_BUILD WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(${legacy} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
  target_link_options(${legacy} PRIVATE "/PDBALTPATH:%_PDB%")
  set_target_properties(${legacy} PROPERTIES OUTPUT_NAME "${target}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/legacy")
endforeach()

target_compile_definitions(azs_lobby_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

add_test(NAME lobby_session COMMAND azs_lobby_tests)
add_executable(azs_lobby_ui_tests tests/lobby_ui_tests.cpp)
target_compile_options(azs_lobby_ui_tests PRIVATE /utf-8)
target_compile_definitions(azs_lobby_ui_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(azs_lobby_ui_tests PRIVATE minhook ws2_32 bcrypt user32 gdi32)
add_test(NAME lobby_ui COMMAND azs_lobby_ui_tests "${CMAKE_SOURCE_DIR}/runtime/lobby-ui")
add_executable(azs_crash_fixture tests/crash_fixture.cpp)
